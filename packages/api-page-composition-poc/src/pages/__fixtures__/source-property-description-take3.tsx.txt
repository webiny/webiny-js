export default function PropertyDescription({ inputs: { title, sectionLabel, description, amenitiesHeading, images, cta } }) {
    const items = images && images.length ? images : [];
    const [active, setActive] = React.useState(0);

    React.useEffect(() => {
        if (items.length < 2) return;
        const interval = setInterval(() => {
            setActive(prev => (prev + 1) % items.length);
        }, 4000);
        return () => clearInterval(interval);
    }, [items.length]);

    return (
        <section className="property-description">
            <div className="property-description-carousel">
                {items.map((image, index) => (
                    image && image.src && image.src.url ? (
                        <img
                            key={index}
                            src={image.src.url}
                            alt={image.alt || ""}
                            className={`property-description-slide${index === active ? " is-active" : ""}`}
                        />
                    ) : null
                ))}
            </div>
            <div className="property-description-content">
                {sectionLabel ? <p className="property-description-label">{sectionLabel}</p> : null}
                <h2 className="property-description-title">{title}</h2>
                {description ? (
                    <div className="property-description-text" dangerouslySetInnerHTML={{ __html: description }} />
                ) : null}
                <p className="property-description-amenities-text">
                    {amenitiesHeading ? (
                        cta && cta.href ? (
                            <a href={cta.href} className="property-description-amenities-link">{amenitiesHeading}</a>
                        ) : (
                            <strong className="property-description-amenities-heading">{amenitiesHeading}</strong>
                        )
                    ) : null}
                    {" "}include a combined laundry and lounge area, a fitness center, and controlled access. Park City Place offers a convenient lifestyle for those seeking the best of urban living in Bridgeport. A parking garage is also available for residents at a monthly charge.
                </p>
            </div>
        </section>
    );
}
