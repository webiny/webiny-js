export default function HeroBanner({ inputs: { eyebrowText, title, cta, images } }) {
    const slides = images && images.length ? images : [];
    const [activeIndex, setActiveIndex] = React.useState(0);

    React.useEffect(() => {
        if (slides.length <= 1) return;
        const interval = setInterval(() => {
            setActiveIndex((prev) => (prev + 1) % slides.length);
        }, 6000);
        return () => clearInterval(interval);
    }, [slides.length]);

    return (
        <section className="hero-banner">
            <div className="hero-banner-images">
                {slides.map((img, index) => (
                    <div
                        key={index}
                        className={`hero-banner-image ${index === activeIndex ? "is-active" : ""}`}
                        style={{ backgroundImage: img && img.src ? `url(${img.src})` : undefined }}
                        role="img"
                        aria-label={img && img.alt ? img.alt : ""}
                    />
                ))}
                <div className="hero-banner-overlay" />
            </div>
            <div className="hero-banner-content">
                {eyebrowText ? <p className="hero-banner-eyebrow">{eyebrowText}</p> : null}
                <h1 className="hero-banner-title">{title}</h1>
                {cta && cta.label ? (
                    <a href={cta.url || "#"} className="hero-banner-cta">{cta.label}</a>
                ) : null}
            </div>
        </section>
    );
}
