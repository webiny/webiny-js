export default function PropertyHighlightsLinks({ inputs: { items } }) {
    const list = items || [];

    return (
        <section className="phl">
            <div className="phl-grid">
                {list.map((item, index) => (
                    <div className="phl-item" key={index}>
                        {item.image && item.image.src ? (
                            <div className="phl-image-wrap">
                                <img
                                    className="phl-image"
                                    src={item.image.src}
                                    alt={item.image.alt || item.heading || ""}
                                />
                            </div>
                        ) : null}
                        {item.heading ? <h3 className="phl-heading">{item.heading}</h3> : null}
                        {item.link && item.link.label ? (
                            <a className="phl-link" href={item.link.url || "#"}>
                                {item.link.label}
                            </a>
                        ) : null}
                    </div>
                ))}
            </div>
        </section>
    );
}
